#!/usr/bin/env bash
# Ray's own account of SI's cluster at one point (StochasticInstanton rrk-adoption prompt 04; the
# orchestrator's brief, correction 11): live workers, alive actors, and every job. Read-only `ray
# list` calls, each addressed explicitly; run from the repository root.
#     bash ray_state.sh LABEL
A=127.0.0.1:6390
echo "== ray state: $1, at $(date '+%Y-%m-%d %H:%M:%S %Z')"
echo "\$ ./venv/bin/ray list workers --address $A --filter \"is_alive=True\" --limit 500"
./venv/bin/ray list workers --address $A --filter "is_alive=True" --limit 500 2>&1; echo "[exit $?]"
echo "\$ ./venv/bin/ray list actors --address $A --filter \"state=ALIVE\""
./venv/bin/ray list actors --address $A --filter "state=ALIVE" 2>&1; echo "[exit $?]"
echo "\$ ./venv/bin/ray list jobs --address $A"
./venv/bin/ray list jobs --address $A 2>&1; echo "[exit $?]"
