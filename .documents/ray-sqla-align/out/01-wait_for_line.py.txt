"""
Wait until LOG holds a line containing TEXT (StochasticInstanton rrk-adoption prompt 04; the
orchestrator's brief, correction 6). Reads LOG only: no pgrep, no kill -0, no Ray, and nothing is
started or signalled.

    ./venv/bin/python wait_for_line.py LOG TEXT --timeout SECONDS [--echo] [--poll SECONDS]

Prints the first line containing TEXT with the time it was first seen (UTC, to 0.1 s) and exits
0; at the deadline it says it timed out and exits 1. A LOG that does not yet exist is read as
empty. With --echo it also prints every line of LOG with the time it was first seen; the lines
already in LOG at the first read are marked "present at start", so their times are upper bounds.
The poll is at most 5 Hz (--poll is clamped to at least 0.2 s).
"""

import argparse
import sys
import time
from datetime import datetime, timezone
from pathlib import Path


def _now() -> str:
    t = datetime.now(timezone.utc)
    return t.strftime("%Y-%m-%dT%H:%M:%S.") + f"{t.microsecond // 100000}Z"


def _lines(log: Path):
    """The complete lines of LOG: a last line with no newline yet is left for the next read."""
    try:
        text = log.read_text(errors="replace")
    except OSError:
        return []
    lines = text.splitlines()
    if lines and not text.endswith("\n"):
        lines.pop()
    return lines


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Wait for a line containing TEXT in LOG.")
    parser.add_argument("log", type=Path)
    parser.add_argument("text")
    parser.add_argument("--timeout", type=float, required=True)
    parser.add_argument("--poll", type=float, default=0.2)
    parser.add_argument("--echo", action="store_true")
    args = parser.parse_args(argv)
    poll = max(args.poll, 0.2)

    start = time.monotonic()
    deadline = start + args.timeout
    seen = 0
    first = True
    print(f"[{_now()}] wait_for_line: waiting for \"{args.text}\" in {args.log}, "
          f"timeout {args.timeout:g} s", flush=True)
    while True:
        lines = _lines(args.log)
        complete = len(lines)
        stamp = _now()
        for i in range(seen, complete):
            line = lines[i]
            if args.echo:
                mark = " (present at start)" if first else ""
                print(f"[{stamp}]{mark} {i + 1:4d}| {line}", flush=True)
            if args.text in line:
                print(f"[{stamp}] wait_for_line: after {time.monotonic() - start:.1f} s, "
                      f"line {i + 1}: {line}", flush=True)
                return 0
        seen = max(seen, complete)
        first = False
        if time.monotonic() >= deadline:
            print(f"[{_now()}] wait_for_line: timed out after {args.timeout:g} s waiting for "
                  f"\"{args.text}\" in {args.log}", flush=True)
            return 1
        time.sleep(min(poll, max(deadline - time.monotonic(), 0.0)))


if __name__ == "__main__":
    sys.exit(main())
